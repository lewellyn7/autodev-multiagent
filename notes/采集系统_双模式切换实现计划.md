> 目标：支持自用模式（免登录）与团队模式（完整认证）动态切换

## 🛠️ 技术实现细节

### 配置层 (`app/config/settings.py`)
```python
class Settings(BaseSettings):
    # ... 现有配置 ...
    
    deployment_mode: str = Field(
        default="team",
        description="部署模式：self(自用) | team(团队)"
    )
    
    @validator("deployment_mode")
    def validate_mode(cls, v):
        if v not in ["self", "team"]:
            raise ValueError("DEPLOYMENT_MODE must be 'self' or 'team'")
        return v
    
    @property
    def is_self_mode(self) -> bool:
        return self.deployment_mode == "self"
```

### 数据库初始化 (`app/database/db.py`)
```python
async def init_self_mode_admin_user():
    """自用模式启动时自动创建 admin 用户"""
    settings = get_settings()
    if not settings.is_self_mode:
        return
    
    # 检查 admin 用户是否存在
    conn = await get_db_conn()
    exists = await conn.fetchval(
        "SELECT 1 FROM users WHERE username = 'admin' LIMIT 1"
    )
    
    if not exists:
        password_hash = bcrypt.hashpw("admin123".encode('utf-8'), bcrypt.gensalt())
        await conn.execute(
            """INSERT INTO users (username, password_hash, role, display_name, created_at)
               VALUES ('admin', $1, 'admin', '系统管理员', NOW())""",
            password_hash.decode('utf-8')
        )
        logger.info("自用模式：已创建默认 admin 用户")
```

### 模式切换 API (`app/api/routes/system.py`)
```python
@router.get("/system/mode")
async def get_mode(current_user: User = Depends(get_current_user)):
    """获取当前部署模式"""
    return {"mode": get_settings().deployment_mode}

@router.post("/system/mode")
async def switch_mode(
    mode: str,
    current_user: User = Depends(get_current_user)
):
    """切换部署模式（需重启生效）"""
    if current_user.role != "admin":
        raise HTTPException(403, "仅管理员可切换模式")
    
    if mode not in ["self", "team"]:
        raise HTTPException(400, "模式必须为 self 或 team")
    
    # 写入配置文件或环境变量
    update_deployment_mode(mode)
    
    return {
        "mode": mode,
        "message": "模式已切换，需重启服务生效"
    }
```
